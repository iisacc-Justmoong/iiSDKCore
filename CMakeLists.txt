cmake_minimum_required(VERSION 3.31)
project(iiSDKCore VERSION 0.2.0 LANGUAGES CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
option(BUILD_SHARED_LIBS "Build iiSDKCore as a shared library" ON)
option(IISDKCORE_BUILD_EXAMPLES "Build runtime examples" ON)
option(IISDKCORE_WASM_THREADS "Enable Emscripten pthread workers (requires a compatible host)" ON)
if(EMSCRIPTEN OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(BUILD_SHARED_LIBS OFF)
endif()
find_package(Threads REQUIRED)

configure_file(src/library.h "${CMAKE_CURRENT_BINARY_DIR}/include/iiSDKCore/library.h" COPYONLY)
set(_runtime_headers Runtime/EventLoop.h Runtime/Stream.h Runtime/BrowserLoop.h Runtime/Query.h Runtime/ObjectProperties.h Runtime/Thread.h Runtime/Runtime.h Runtime/Value.h IO/Channel.h IO/AsyncIO.h)
include(cmake/Foundations.cmake)
list(APPEND _runtime_headers ${_foundation_headers})
foreach(_header IN LISTS _runtime_headers)
    configure_file("src/${_header}" "${CMAKE_CURRENT_BINARY_DIR}/include/iiSDKCore/${_header}" COPYONLY)
endforeach()
add_library(iiSDKCore src/library.cpp src/Runtime/EventLoop.cpp src/Runtime/ObjectProperties.cpp src/Runtime/Thread.cpp src/Runtime/Runtime.cpp src/Runtime/Value.cpp src/IO/Channel.cpp src/IO/AsyncIO.cpp)
target_sources(iiSDKCore PRIVATE ${_foundation_sources})
if(EMSCRIPTEN)
    target_sources(iiSDKCore PRIVATE src/Runtime/BrowserLoop.cpp)
    # Callback failures must retain the same C++ exception semantics as native builds.
    target_compile_options(iiSDKCore PUBLIC -fexceptions)
    target_link_options(iiSDKCore PUBLIC -fexceptions)
    if(IISDKCORE_WASM_THREADS)
        target_compile_options(iiSDKCore PUBLIC -pthread)
        target_link_options(iiSDKCore PUBLIC -pthread)
    endif()
endif()
add_library(iiSDKCore::iiSDKCore ALIAS iiSDKCore)
target_compile_features(iiSDKCore PUBLIC cxx_std_23)
target_link_libraries(iiSDKCore PUBLIC Threads::Threads)
target_include_directories(iiSDKCore PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/iiSDKCore>)
set_target_properties(iiSDKCore PROPERTIES
    CXX_EXTENSIONS OFF
    VERSION ${PROJECT_VERSION}
    SOVERSION ${PROJECT_VERSION_MAJOR}
    WINDOWS_EXPORT_ALL_SYMBOLS ON)

set(_package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/iiSDKCore")
install(TARGETS iiSDKCore EXPORT iiSDKCoreTargets
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES src/library.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiSDKCore)
foreach(_header IN LISTS _runtime_headers)
    get_filename_component(_directory "${_header}" DIRECTORY)
    install(FILES "src/${_header}" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/iiSDKCore/${_directory}")
endforeach()
install(EXPORT iiSDKCoreTargets NAMESPACE iiSDKCore:: DESTINATION ${_package_dir})
configure_package_config_file(cmake/iiSDKCoreConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/iiSDKCoreConfig.cmake"
    INSTALL_DESTINATION ${_package_dir})
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiSDKCoreConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES
    "${CMAKE_CURRENT_BINARY_DIR}/iiSDKCoreConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/iiSDKCoreConfigVersion.cmake"
    DESTINATION ${_package_dir})

if(BUILD_TESTING)
    # Each public foundation header must compile without accidental include order.
    set(_header_checks)
    foreach(_header IN LISTS _foundation_headers)
        string(MAKE_C_IDENTIFIER "${_header}" _header_id)
        set(_check "${CMAKE_CURRENT_BINARY_DIR}/header-checks/${_header_id}.cpp")
        file(WRITE "${_check}" "#include <iiSDKCore/${_header}>\n")
        list(APPEND _header_checks "${_check}")
    endforeach()
    add_library(iiSDKCorePublicHeaders OBJECT ${_header_checks})
    target_link_libraries(iiSDKCorePublicHeaders PRIVATE iiSDKCore::iiSDKCore)
    add_executable(iiSDKCoreFoundationsTests tests/foundations.cpp)
    target_link_libraries(iiSDKCoreFoundationsTests PRIVATE iiSDKCore::iiSDKCore)
    add_test(NAME iiSDKCore.Foundations COMMAND iiSDKCoreFoundationsTests)
    set_tests_properties(iiSDKCore.Foundations PROPERTIES TIMEOUT 30)
    add_executable(iiSDKCoreIOTests tests/io.cpp)
    target_link_libraries(iiSDKCoreIOTests PRIVATE iiSDKCore::iiSDKCore)
    add_test(NAME iiSDKCore.IO COMMAND iiSDKCoreIOTests)
    set_tests_properties(iiSDKCore.IO PROPERTIES TIMEOUT 30)
    add_executable(iiSDKCoreValueTests tests/value.cpp)
    target_link_libraries(iiSDKCoreValueTests PRIVATE iiSDKCore::iiSDKCore)
    add_test(NAME iiSDKCore.Value COMMAND iiSDKCoreValueTests)
    set_tests_properties(iiSDKCore.Value PROPERTIES TIMEOUT 30)
    add_executable(iiSDKCoreParallelTests tests/parallel.cpp)
    target_link_libraries(iiSDKCoreParallelTests PRIVATE iiSDKCore::iiSDKCore)
    add_test(NAME iiSDKCore.Parallel COMMAND iiSDKCoreParallelTests)
    set_tests_properties(iiSDKCore.Parallel PROPERTIES TIMEOUT 30)
    add_executable(iiSDKCoreThreadTests tests/thread.cpp)
    target_link_libraries(iiSDKCoreThreadTests PRIVATE iiSDKCore::iiSDKCore)
    add_test(NAME iiSDKCore.Thread COMMAND iiSDKCoreThreadTests)

    add_executable(iiSDKCoreMemberTreeTests tests/member_tree.cpp)
    target_link_libraries(iiSDKCoreMemberTreeTests PRIVATE iiSDKCore::iiSDKCore)
    add_test(NAME iiSDKCore.MemberTree COMMAND iiSDKCoreMemberTreeTests)
    set_tests_properties(iiSDKCore.MemberTree PROPERTIES TIMEOUT 20)
    add_executable(iiSDKCorePropertiesTests tests/properties.cpp)
    target_link_libraries(iiSDKCorePropertiesTests PRIVATE iiSDKCore::iiSDKCore)
    add_test(NAME iiSDKCore.Properties COMMAND iiSDKCorePropertiesTests)
    set_tests_properties(iiSDKCore.Properties PROPERTIES TIMEOUT 20)
    # Native allocation instrumentation; WASM malloc is managed by its host runtime.
    if(NOT EMSCRIPTEN)
        add_executable(iiSDKCoreReuseTests tests/reuse.cpp)
        target_link_libraries(iiSDKCoreReuseTests PRIVATE iiSDKCore::iiSDKCore)
        add_test(NAME iiSDKCore.Reuse COMMAND iiSDKCoreReuseTests)
        set_tests_properties(iiSDKCore.Reuse PROPERTIES TIMEOUT 20)
    else()
        add_executable(iiSDKCoreBrowserTests tests/browser.cpp)
        target_link_libraries(iiSDKCoreBrowserTests PRIVATE iiSDKCore::iiSDKCore)
        set_target_properties(iiSDKCoreBrowserTests PROPERTIES SUFFIX ".html")
        target_link_options(iiSDKCoreBrowserTests PRIVATE "-sEXIT_RUNTIME=0")

# Only programs that own/join workers move main() off the browser host thread.
# GUI/BrowserLoop consumers keep their host-thread entry point.
if(EMSCRIPTEN AND IISDKCORE_WASM_THREADS)
    if(BUILD_TESTING)
        set(IISDKCORE_BROWSER_CONTRACT_TITLE "iiSDKCore foundation contracts")
        set(IISDKCORE_BROWSER_CONTRACT_SCRIPT "iiSDKCoreFoundationsTests.js")
        configure_file(tests/thread-browser.html "${CMAKE_CURRENT_BINARY_DIR}/foundations-browser.html" @ONLY)
        set(IISDKCORE_BROWSER_CONTRACT_TITLE "iiSDKCore worker thread contracts")
        set(IISDKCORE_BROWSER_CONTRACT_SCRIPT "iiSDKCoreThreadTests.js")
        configure_file(tests/thread-browser.html "${CMAKE_CURRENT_BINARY_DIR}/thread-browser.html" @ONLY)
        set(IISDKCORE_BROWSER_CONTRACT_TITLE "iiSDKCore parallel runtime contracts")
        set(IISDKCORE_BROWSER_CONTRACT_SCRIPT "iiSDKCoreParallelTests.js")
        configure_file(tests/thread-browser.html "${CMAKE_CURRENT_BINARY_DIR}/parallel-browser.html" @ONLY)
        set(IISDKCORE_BROWSER_CONTRACT_TITLE "iiSDKCore dynamic value contracts")
        set(IISDKCORE_BROWSER_CONTRACT_SCRIPT "iiSDKCoreValueTests.js")
        configure_file(tests/thread-browser.html "${CMAKE_CURRENT_BINARY_DIR}/value-browser.html" @ONLY)
        set(IISDKCORE_BROWSER_CONTRACT_TITLE "iiSDKCore I/O contracts")
        set(IISDKCORE_BROWSER_CONTRACT_SCRIPT "iiSDKCoreIOTests.js")
        configure_file(tests/thread-browser.html "${CMAKE_CURRENT_BINARY_DIR}/io-browser.html" @ONLY)
    endif()
    foreach(_worker_program iiSDKCoreFoundationsTests iiSDKCoreFoundationsExample iiSDKCoreIOTests iiSDKCoreValueTests iiSDKCoreParallelTests iiSDKCoreParallelExample iiSDKCoreThreadTests iiSDKCoreRuntimeTests iiSDKCoreQueryTests iiSDKCoreThreadExample)
        if(TARGET ${_worker_program})
            target_link_options(${_worker_program} PRIVATE
                "-sPROXY_TO_PTHREAD=1" "-sPTHREAD_POOL_SIZE=16" "-sEXIT_RUNTIME=1"
                "-sALLOW_BLOCKING_ON_MAIN_THREAD=0")
        endif()
    endforeach()
endif()
