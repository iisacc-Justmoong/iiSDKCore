    if(IISDKCORE_STATIC_PYTHON_SOURCES)
        if(NOT EXISTS "${IISDKCORE_STATIC_PYTHON_COMPILER}")
            message(FATAL_ERROR "Native Python source compiler is unavailable")
        endif()
        foreach(_python_source IN LISTS IISDKCORE_STATIC_PYTHON_SOURCES)
            get_filename_component(_python_source "${_python_source}" ABSOLUTE)
            string(SHA256 _python_hash "${_python_source}")
            string(SUBSTRING "${_python_hash}" 0 24 _python_hash)
            execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${IISDKCORE_STATIC_PYTHON_COMPILER}"
                --source "${_python_source}" RESULT_VARIABLE _python_result
                OUTPUT_VARIABLE _python_cpp ERROR_VARIABLE _python_error ENCODING UTF-8)
            if(NOT _python_result STREQUAL "0")
                message(FATAL_ERROR "Native Python compilation failed: ${_python_error}")
            endif()
            set(_python_generated "${CMAKE_CURRENT_BINARY_DIR}/iiSDKCorePython_${_python_hash}.cpp")
            file(GENERATE OUTPUT "${_python_generated}" CONTENT "${_python_cpp}")
            set_source_files_properties("${_python_generated}" PROPERTIES GENERATED TRUE)
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
                "${_python_source}" "${IISDKCORE_STATIC_PYTHON_COMPILER}")
            list(APPEND _static_plugin_sources "${_python_generated}")
        endforeach()
    endif()
    foreach(_plugin_source IN LISTS _static_plugin_sources)
        get_filename_component(_plugin_source "${_plugin_source}" ABSOLUTE)
        get_source_file_property(_generated "${_plugin_source}" GENERATED)
        if((NOT EXISTS "${_plugin_source}" AND NOT _generated) OR IS_DIRECTORY "${_plugin_source}")
            message(FATAL_ERROR "Static plugin source is missing: ${_plugin_source}")
        endif()
        if(_plugin_source MATCHES "\\.mm$" AND APPLE)
            enable_language(OBJCXX)
        endif()
        string(SHA256 _plugin_hash "${_plugin_source}")
        string(SUBSTRING "${_plugin_hash}" 0 24 _plugin_hash)
        set(_plugin_symbol "ii_sdk_static_plugin_${_plugin_hash}")
        set_source_files_properties("${_plugin_source}" PROPERTIES COMPILE_DEFINITIONS "ii_sdk_plugin_v1=${_plugin_symbol}")
        target_sources(iiSDKCore PRIVATE "${_plugin_source}")
        string(APPEND _static_declarations "extern \"C\" const ii::plugin::Descriptor* ${_plugin_symbol}() noexcept;\n")
        string(APPEND _static_entries "  &${_plugin_symbol},\n")
        math(EXPR _static_count "${_static_count}+1")
    endforeach()
    set(_static_catalog "${CMAKE_CURRENT_BINARY_DIR}/iiSDKCoreStaticPlugins.cpp")
    file(GENERATE OUTPUT "${_static_catalog}" CONTENT
        "#include <iiSDKCore/Plugin/Plugin.h>\n#include <array>\n#include <span>\n${_static_declarations}namespace ii::plugin::detail {\nstd::span<const Entry> embedded_entries() noexcept {\nstatic constexpr std::array<Entry, ${_static_count}> entries{\n${_static_entries}};\nreturn entries;\n}\n}\n")
    target_sources(iiSDKCore PRIVATE "${_static_catalog}")
    if(IISDKCORE_STATIC_PLUGIN_INCLUDE_DIRS)
        target_include_directories(iiSDKCore PRIVATE ${IISDKCORE_STATIC_PLUGIN_INCLUDE_DIRS})
    endif()
