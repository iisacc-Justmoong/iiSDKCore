    check_cxx_symbol_exists(POSIX_SPAWN_CLOEXEC_DEFAULT "spawn.h" _process_cloexec)

        target_compile_definitions(iiSDKCore PRIVATE IISDKCORE_PROCESS_POSIX=1)

            target_compile_definitions(iiSDKCore PRIVATE IISDKCORE_PROCESS_CLOSEFROM=1 _GNU_SOURCE)

    endif()
