cmake_minimum_required(VERSION 3.31)
project(iiSDKCore VERSION 0.10.0 LANGUAGES CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
include(cmake/ArchitecturePolicy.cmake)
option(BUILD_SHARED_LIBS "Build iiSDKCore as a shared library" ON)
option(IISDKCORE_BUILD_EXAMPLES "Build runtime examples" ON)
option(IISDKCORE_WASM_THREADS "Enable Emscripten pthread workers (requires a compatible host)" ON)
if(EMSCRIPTEN OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(BUILD_SHARED_LIBS OFF)
endif()
find_package(Threads REQUIRED)

configure_file(src/library.h "${CMAKE_CURRENT_BINARY_DIR}/include/iiSDKCore/library.h" COPYONLY)
set(_runtime_headers Runtime/EventLoop.h Runtime/Stream.h Runtime/LiveStream.h Runtime/BrowserLoop.h Runtime/Query.h Runtime/TaskScope.h Runtime/ObjectProperties.h Runtime/Thread.h Runtime/Runtime.h Runtime/Value.h Runtime/Serialization.h IO/Channel.h IO/Requests.h IO/AsyncIO.h IO/Hardware.h IO/Protocols.h Runtime/Animation.h IO/AtomicFile.h Plugin/Plugin.h)
include(cmake/Foundations.cmake)
list(APPEND _runtime_headers ${_foundation_headers})
foreach(_header IN LISTS _runtime_headers)
    configure_file("src/${_header}" "${CMAKE_CURRENT_BINARY_DIR}/include/iiSDKCore/${_header}" COPYONLY)
endforeach()
add_library(iiSDKCore src/library.cpp src/Runtime/EventLoop.cpp src/Runtime/TaskScope.cpp src/Runtime/ObjectProperties.cpp src/Runtime/Thread.cpp src/Runtime/Runtime.cpp src/Runtime/Value.cpp src/IO/Channel.cpp src/IO/AsyncIO.cpp)
target_sources(iiSDKCore PRIVATE src/IO/Hardware.cpp src/IO/Protocols.cpp src/Runtime/Animation.cpp src/IO/AtomicFile.cpp src/Plugin/Plugin.cpp)
target_link_libraries(iiSDKCore PRIVATE ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_compile_definitions(iiSDKCore PRIVATE IISDKCORE_STATIC_PLUGINS=1)
endif()
set(IISDKCORE_PACKAGE_LINUX_IO OFF)
if(APPLE)
    enable_language(OBJCXX)

    target_sources(iiSDKCore PRIVATE src/IO/HardwareApple.mm)
    set_source_files_properties(src/IO/HardwareApple.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc;-fblocks")
    foreach(_framework Foundation CoreBluetooth AudioToolbox CoreMIDI)
        find_library(_hw_${_framework} ${_framework} REQUIRED)
        target_link_libraries(iiSDKCore PRIVATE ${_hw_${_framework}})
    endforeach()
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        find_library(_hw_AVFAudio AVFAudio REQUIRED)
        target_link_libraries(iiSDKCore PRIVATE ${_hw_AVFAudio})
    else()
        foreach(_framework IOKit IOUSBHost CoreAudio)
            find_library(_hw_${_framework} ${_framework} REQUIRED)
            target_link_libraries(iiSDKCore PRIVATE ${_hw_${_framework}})
        endforeach()
    endif()
elseif(WIN32)
    target_sources(iiSDKCore PRIVATE src/IO/HardwareWindows.cpp)
    target_compile_definitions(iiSDKCore PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    target_link_libraries(iiSDKCore PRIVATE setupapi cfgmgr32 hid winusb ole32 uuid runtimeobject)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(IISDKCORE_LINUX_IO REQUIRED IMPORTED_TARGET libudev dbus-1 alsa>=1.2.10)
    target_sources(iiSDKCore PRIVATE src/IO/HardwareLinux.cpp)
    target_link_libraries(iiSDKCore PRIVATE PkgConfig::IISDKCORE_LINUX_IO)
    set(IISDKCORE_PACKAGE_LINUX_IO ON)
else()
    target_sources(iiSDKCore PRIVATE src/IO/HardwarePortable.cpp)
endif()
target_sources(iiSDKCore PRIVATE ${_foundation_sources})
if(EMSCRIPTEN)
    target_sources(iiSDKCore PRIVATE src/Runtime/BrowserLoop.cpp)
    # Callback failures must retain the same C++ exception semantics as native builds.
    target_compile_options(iiSDKCore PUBLIC -fexceptions)
    target_link_options(iiSDKCore PUBLIC -fexceptions)
    if(IISDKCORE_WASM_THREADS)
        target_compile_options(iiSDKCore PUBLIC -pthread)
        target_link_options(iiSDKCore PUBLIC -pthread)
    endif()
endif()
add_library(iiSDKCore::iiSDKCore ALIAS iiSDKCore)
# Lower value modules can be consumed without loading the runtime or OS I/O.
add_library(iiSDKCoreValues INTERFACE)
add_library(iiSDKCore::Values ALIAS iiSDKCoreValues)
set_target_properties(iiSDKCoreValues PROPERTIES EXPORT_NAME Values)
target_compile_features(iiSDKCoreValues INTERFACE cxx_std_23)
# Value-only WASM consumers retain the same exception contract, including when
# consuming an installed header package produced on another platform.
target_compile_options(iiSDKCoreValues INTERFACE "$<$<PLATFORM_ID:Emscripten>:-fexceptions>")
target_link_options(iiSDKCoreValues INTERFACE "$<$<PLATFORM_ID:Emscripten>:-fexceptions>")
target_include_directories(iiSDKCoreValues INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

target_compile_features(iiSDKCore PUBLIC cxx_std_23)
target_link_libraries(iiSDKCore PUBLIC Threads::Threads)
target_include_directories(iiSDKCore PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/iiSDKCore>)
set_target_properties(iiSDKCore PROPERTIES
    CXX_EXTENSIONS OFF
    VERSION ${PROJECT_VERSION}
    SOVERSION ${PROJECT_VERSION_MAJOR}
    WINDOWS_EXPORT_ALL_SYMBOLS ON)

set(_package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/iiSDKCore")
install(TARGETS iiSDKCore EXPORT iiSDKCoreTargets
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES src/library.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiSDKCore)
foreach(_header IN LISTS _runtime_headers)
    get_filename_component(_directory "${_header}" DIRECTORY)
    install(FILES "src/${_header}" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/iiSDKCore/${_directory}")
endforeach()
install(TARGETS iiSDKCoreValues EXPORT iiSDKCoreValuesTargets)
install(EXPORT iiSDKCoreValuesTargets NAMESPACE iiSDKCore:: DESTINATION ${_package_dir})
install(EXPORT iiSDKCoreTargets NAMESPACE iiSDKCore:: DESTINATION ${_package_dir})
configure_package_config_file(cmake/iiSDKCoreConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/iiSDKCoreConfig.cmake"
    INSTALL_DESTINATION ${_package_dir})
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiSDKCoreVersionPolicy.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion ARCH_INDEPENDENT)
math(EXPR IISDKCORE_PACKAGE_BITS "${CMAKE_SIZEOF_VOID_P} * 8")
configure_file(cmake/iiSDKCoreConfigVersion.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/iiSDKCoreConfigVersion.cmake" @ONLY)
install(FILES
    "${CMAKE_CURRENT_BINARY_DIR}/iiSDKCoreConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/iiSDKCoreConfigVersion.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/iiSDKCoreVersionPolicy.cmake"
    DESTINATION ${_package_dir})
