cmake_minimum_required(VERSION 3.31)
project(TrackedOperationConsumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderNative TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderCoreIO TRUE)
find_package(iiSDKCore 0.14 CONFIG REQUIRED COMPONENTS Runtime)
add_executable(consumer ../operation.cpp)
target_link_libraries(consumer PRIVATE iiSDKCore::Runtime)
set_target_properties(consumer PROPERTIES CXX_EXTENSIONS OFF)
if(EMSCRIPTEN)
    target_link_options(consumer PRIVATE "-sPROXY_TO_PTHREAD=1" "-sPTHREAD_POOL_SIZE=16" "-sEXIT_RUNTIME=1" "-sALLOW_BLOCKING_ON_MAIN_THREAD=0")
endif()
enable_testing()
add_test(NAME iiSDKCore.InstalledOperation COMMAND consumer)
set_tests_properties(iiSDKCore.InstalledOperation PROPERTIES TIMEOUT 30)
