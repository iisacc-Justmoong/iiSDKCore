cmake_minimum_required(VERSION 3.31)
project(iiSDKCoreIndependentContracts LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Threads TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderNative TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderCoreIO TRUE)
find_package(iiSDKCore 0.14 CONFIG REQUIRED COMPONENTS Contracts)
if(TARGET iiSDKCore::iiSDKCore OR TARGET iiSDKCore::Runtime OR TARGET iiSDKCore::Values
    OR TARGET Qt6::Core OR TARGET Threads::Threads OR TARGET iiFileProvider::Native OR TARGET iiFileProvider::CoreIO)
    message(FATAL_ERROR "Contract import pulled an implementation dependency")
endif()
add_executable(iiSDKCoreIndependentContracts ../contracts.cpp)
target_link_libraries(iiSDKCoreIndependentContracts PRIVATE iiSDKCore::Contracts)
enable_testing()
add_test(NAME iiSDKCore.IndependentContracts COMMAND iiSDKCoreIndependentContracts)
add_executable(iiSDKCoreIndependentText ../text.cpp)
target_link_libraries(iiSDKCoreIndependentText PRIVATE iiSDKCore::Contracts)
add_test(NAME iiSDKCore.IndependentText COMMAND iiSDKCoreIndependentText)
