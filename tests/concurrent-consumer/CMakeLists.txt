cmake_minimum_required(VERSION 3.31)
project(ConcurrentConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderNative TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderCoreIO TRUE)
find_package(iiSDKCore CONFIG REQUIRED COMPONENTS Runtime)
enable_testing()
foreach(_family future concurrent)
    add_executable(${_family} ../${_family}.cpp)
    target_link_libraries(${_family} PRIVATE iiSDKCore::Runtime)
    if(EMSCRIPTEN)
        target_link_options(${_family} PRIVATE "-sPROXY_TO_PTHREAD=1" "-sPTHREAD_POOL_SIZE=16" "-sEXIT_RUNTIME=1" "-sALLOW_BLOCKING_ON_MAIN_THREAD=0")
    endif()
    add_test(NAME iiSDKCore.Installed_${_family} COMMAND ${_family})
    set_tests_properties(iiSDKCore.Installed_${_family} PROPERTIES TIMEOUT 30)
endforeach()
