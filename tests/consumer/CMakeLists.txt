cmake_minimum_required(VERSION 3.31)
project(iiSDKCoreInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8)
endif()

set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderNative TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderCoreIO TRUE)
find_package(iiSDKCore 0.20 CONFIG REQUIRED COMPONENTS Runtime Values)
if(TARGET Qt6::Core OR TARGET iiFileProvider::Native OR TARGET iiFileProvider::CoreIO
    OR TARGET iiDeviceMultimedia::iiDeviceMultimedia OR TARGET PkgConfig::IIDEVICEMULTIMEDIA_LINUX_IO)
    message(FATAL_ERROR "Core-only package import pulled an upper device module")
endif()
get_target_property(_core_links iiSDKCore::iiSDKCore INTERFACE_LINK_LIBRARIES)
if(_core_links MATCHES "iiFileProvider|Qt[56]|iiDeviceMultimedia|PkgConfig|CoreBluetooth|CoreMIDI|AudioToolbox|IOUSBHost")
    message(FATAL_ERROR "Core-only target exposes device/media dependencies")
endif()
get_target_property(_core_includes iiSDKCore::iiSDKCore INTERFACE_INCLUDE_DIRECTORIES)
foreach(_include IN LISTS _core_includes)
    if(EXISTS "${_include}/iiSDKCore/IO/Hardware.h" OR EXISTS "${_include}/iiSDKCore/IO/Protocols.h"
        OR EXISTS "${_include}/iiSDKCore/IO/AtomicFile.h")
        message(FATAL_ERROR "Obsolete Core device headers survived package upgrade")
    endif()
endforeach()
