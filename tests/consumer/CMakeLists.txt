cmake_minimum_required(VERSION 3.31)
project(iiSDKCoreInstalledConsumer LANGUAGES CXX)

find_package(iiSDKCore 0.10 CONFIG REQUIRED)
add_executable(iiSDKCoreInstalledConsumer main.cpp)
target_link_libraries(iiSDKCoreInstalledConsumer PRIVATE iiSDKCore::iiSDKCore)
enable_testing()
add_test(NAME iiSDKCore.InstalledConsumer COMMAND iiSDKCoreInstalledConsumer)
set_tests_properties(iiSDKCore.InstalledConsumer PROPERTIES PASS_REGULAR_EXPRESSION "Hello, World!")
add_executable(iiSDKCoreInstalledFoundations ../foundations.cpp)
target_link_libraries(iiSDKCoreInstalledFoundations PRIVATE iiSDKCore::iiSDKCore)
add_test(NAME iiSDKCore.InstalledFoundations COMMAND iiSDKCoreInstalledFoundations)
set_tests_properties(iiSDKCore.InstalledFoundations PROPERTIES TIMEOUT 30)
add_executable(iiSDKCoreInstalledCleanup ../cleanup.cpp)
target_link_libraries(iiSDKCoreInstalledCleanup PRIVATE iiSDKCore::iiSDKCore)
add_test(NAME iiSDKCore.InstalledCleanup COMMAND iiSDKCoreInstalledCleanup)
set_tests_properties(iiSDKCore.InstalledCleanup PROPERTIES TIMEOUT 30)
add_executable(iiSDKCoreInstalledLoopIO ../loop_io.cpp)
target_link_libraries(iiSDKCoreInstalledLoopIO PRIVATE iiSDKCore::iiSDKCore)
add_test(NAME iiSDKCore.InstalledLoopIO COMMAND iiSDKCoreInstalledLoopIO)
set_tests_properties(iiSDKCore.InstalledLoopIO PROPERTIES TIMEOUT 30)

foreach(_test Protocols Hardware Realtime Modern Plugins)
    string(TOLOWER "${_test}" _source)
    add_executable(iiSDKCoreInstalled${_test} ../${_source}.cpp)
    target_link_libraries(iiSDKCoreInstalled${_test} PRIVATE iiSDKCore::iiSDKCore)
    add_test(NAME iiSDKCore.Installed${_test} COMMAND iiSDKCoreInstalled${_test})
    set_tests_properties(iiSDKCore.Installed${_test} PROPERTIES TIMEOUT 30)
endforeach()
if(WIN32)
    target_link_libraries(iiSDKCoreInstalledHardware PRIVATE ole32)
endif()
if(APPLE)
    set_tests_properties(iiSDKCore.InstalledHardware PROPERTIES
        ENVIRONMENT "OBJC_DEBUG_MISSING_POOLS=YES"
        FAIL_REGULAR_EXPRESSION "MISSING POOLS")
endif()

add_executable(iiSDKCoreInstalledValueModules ../value_modules.cpp)
target_link_libraries(iiSDKCoreInstalledValueModules PRIVATE iiSDKCore::Values)
add_test(NAME iiSDKCore.InstalledValueModules COMMAND iiSDKCoreInstalledValueModules)

add_executable(iiSDKCoreInstalledSerialization ../serialization.cpp)
target_link_libraries(iiSDKCoreInstalledSerialization PRIVATE iiSDKCore::Values)
add_test(NAME iiSDKCore.InstalledSerialization COMMAND iiSDKCoreInstalledSerialization)
add_executable(iiSDKCoreInstalledDynamicSerialization ../dynamic_serialization.cpp)
target_link_libraries(iiSDKCoreInstalledDynamicSerialization PRIVATE iiSDKCore::iiSDKCore)
add_test(NAME iiSDKCore.InstalledDynamicSerialization COMMAND iiSDKCoreInstalledDynamicSerialization)

add_executable(iiSDKCoreInstalledSharing ../sharing.cpp)
target_link_libraries(iiSDKCoreInstalledSharing PRIVATE iiSDKCore::Values)
add_test(NAME iiSDKCore.InstalledSharing COMMAND iiSDKCoreInstalledSharing)
