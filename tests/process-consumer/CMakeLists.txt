cmake_minimum_required(VERSION 3.31)
project(NativeProcessConsumer LANGUAGES CXX)
if(MSVC)
    add_compile_options(/utf-8)
endif()
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderNative TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderCoreIO TRUE)
find_package(iiSDKCore 0.15 CONFIG REQUIRED COMPONENTS Runtime)
add_executable(process_fixture ../process_fixture.cpp)
target_link_libraries(process_fixture PRIVATE iiSDKCore::Runtime)
if(WIN32 AND NOT MSVC)
    target_link_options(process_fixture PRIVATE -municode)
endif()
add_executable(consumer ../process.cpp)
target_link_libraries(consumer PRIVATE iiSDKCore::Runtime)
set_target_properties(consumer process_fixture PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME iiSDKCore.InstalledProcess COMMAND consumer $<TARGET_FILE:process_fixture>)
set_tests_properties(iiSDKCore.InstalledProcess PROPERTIES TIMEOUT 30)
