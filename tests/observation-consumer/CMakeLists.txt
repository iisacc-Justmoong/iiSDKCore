cmake_minimum_required(VERSION 3.31)
project(iiSDKCoreIndependentObservation LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Threads TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderNative TRUE)
find_package(iiSDKCore CONFIG REQUIRED COMPONENTS Observation)
if(TARGET iiSDKCore::Runtime OR TARGET iiSDKCore::iiSDKCore OR TARGET iiSDKCore::Values OR TARGET Threads::Threads OR TARGET Qt6::Core)
    message(FATAL_ERROR "Observation imported the application Runtime or a platform service")
endif()
add_library(model SHARED fixtures/observable_module/Model.cpp)
target_compile_definitions(model PRIVATE IISDKCORE_TEST_OBSERVABLE_MODULE=1)
target_link_libraries(model PRIVATE iiSDKCore::Observation)
set_target_properties(model PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE model iiSDKCore::Observation)
if(WIN32)
    add_custom_command(TARGET consumer POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        $<TARGET_RUNTIME_DLLS:consumer> $<TARGET_FILE_DIR:consumer> COMMAND_EXPAND_LISTS)
endif()
enable_testing()
foreach(_case basic nested exceptions threads bounds lifetime)
    add_test(NAME iiSDKCore.ObservationConsumer.${_case} COMMAND consumer ${_case})
endforeach()
