cmake_minimum_required(VERSION 3.31)
project(SelectiveLiveStream LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(iiSDKCore 0.18 CONFIG REQUIRED COMPONENTS Runtime)
if(ANDROID)
    add_library(QueryStreamExample SHARED ../query_stream.cpp)
else()
    add_executable(QueryStreamExample ../query_stream.cpp)
endif()
target_link_libraries(QueryStreamExample PRIVATE iiSDKCore::Runtime)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set_target_properties(QueryStreamExample PROPERTIES MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_GUI_IDENTIFIER com.iisacc.selective-live-stream)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(QueryStreamContracts ../../tests/query_stream.cpp)
    target_link_libraries(QueryStreamContracts PRIVATE iiSDKCore::Runtime)
    if(NOT CMAKE_CROSSCOMPILING)
        add_test(NAME SelectiveLiveStream.Example COMMAND QueryStreamExample)
        add_test(NAME SelectiveLiveStream.Contracts COMMAND QueryStreamContracts)
        set_tests_properties(SelectiveLiveStream.Example SelectiveLiveStream.Contracts PROPERTIES TIMEOUT 30)
    endif()
endif()
if(EMSCRIPTEN)
    foreach(_program QueryStreamExample QueryStreamContracts)
        if(TARGET ${_program})
            target_link_options(${_program} PRIVATE "-sPROXY_TO_PTHREAD=1" "-sPTHREAD_POOL_SIZE=16"
                "-sEXIT_RUNTIME=1" "-sALLOW_BLOCKING_ON_MAIN_THREAD=0")
        endif()
    endforeach()
endif()
